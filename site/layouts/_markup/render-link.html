{{- /* Links in the Markdown are relative paths to other files in the repository, so they work on GitHub. Each page knows its own path in the repository (srcPath, from cmd/docgen), and data/pages.json maps every file the site hosts to its page. On the site: - a link to a file the site hosts goes to its page, never to GitHub; - a link to an image in docs/images/ goes to the site's copy; - a link to any other file (code, scripts, config) goes to its source on GitHub. scripts/check-site-links.py checks the built site for this. */ -}} {{- $dest := .Destination -}} {{- $url := urls.Parse $dest -}} {{- $fragment := cond (ne $url.Fragment "") (printf "#%s" $url.Fragment) "" -}} {{- $local := and $dest (not $url.IsAbs) (not (hasPrefix $dest "#")) (not (hasPrefix $dest "/")) (not (hasPrefix $dest "mailto:")) -}} {{- if $local -}} {{- $src := .PageInner.Params.srcpath | default "" -}} {{- if and (not $src) .PageInner.File -}} {{- $src = path.Join "docs" .PageInner.File.Path -}} {{- end -}} {{- $repoPath := path.Clean (path.Join (path.Dir $src) $url.Path) -}} {{- $resolved := false -}} {{- with index hugo.Data.pages $repoPath -}} {{- with site.GetPage . -}} {{- $dest = printf "%s%s" .RelPermalink $fragment -}} {{- $resolved = true -}} {{- end -}} {{- end -}} {{- if not $resolved -}} {{- if hasPrefix $repoPath "docs/images/" -}} {{- $dest = relURL (strings.TrimPrefix "docs/" $repoPath) -}} {{- else -}} {{- $kind := cond (strings.HasSuffix $url.Path "/") "tree" "blob" -}} {{- $dest = printf "%s/%s/main/%s%s" site.Params.repo $kind $repoPath $fragment -}} {{- end -}} {{- end -}} {{- end -}} {{- $external := strings.HasPrefix $dest "http" -}} {{ .Text | safeHTML }} {{- /* */ -}}
