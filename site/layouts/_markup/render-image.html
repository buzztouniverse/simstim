{{- /* Images in docs/ are relative paths; the site serves docs/images/ at /images/. */ -}} {{- $dest := .Destination -}} {{- $url := urls.Parse $dest -}} {{- if and $dest (not $url.IsAbs) (not (hasPrefix $dest "/")) -}} {{- $src := .PageInner.Params.srcpath | default "" -}} {{- if and (not $src) .PageInner.File -}} {{- $src = path.Join "docs" .PageInner.File.Path -}} {{- end -}} {{- $repoPath := path.Clean (path.Join (path.Dir $src) $url.Path) -}} {{- if hasPrefix $repoPath "docs/images/" -}} {{- $dest = relURL (strings.TrimPrefix "docs/" $repoPath) -}} {{- end -}} {{- end -}} {{- /* */ -}}
